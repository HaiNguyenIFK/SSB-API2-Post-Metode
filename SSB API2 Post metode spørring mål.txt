let
    Url = ".....&outputFormatParams=UseTexts",

    Body =
"
.......
",

    Source =
        Csv.Document(
            Web.Contents(
                Url,
                [
                    Headers = [
                        #"Content-Type" = "application/json"
                    ],
                    Content = Text.ToBinary(Body)
                ]
            ),
            [
                Delimiter = ",",
                Encoding = 1252,
                QuoteStyle = QuoteStyle.Csv
            ]
        )
in
    Source
